        end
        subgraph ERPP["ERP process data"]
            CRIT["Criticality level<br/>for each situation"]:::realdata
            MAT["Welding Material"]:::realdata
        end
    end

    %% PRE-PROCESS
    %% ============================================================

        direction TB
        subgraph BLUR["Blur analysis"]

            BC["Blur Class"]:::software
        end
        subgraph IMGT["Image transformation"]
            HSL["Convert to HSL"]:::software
            UNBLUR["Unblur if needed"]:::trainedaug
            UNROT["Un-Rotate"]:::trainedaug
            RESIZE["Resize, Normalize"]:::software
            FLIP["Flip (mirror effect)<br/>to 3 welding types"]:::software
        end
        subgraph DESC["Image descriptive attributes"]
            SYNCAL["Sync alignement"]:::software
        end
        FBE(("Welder<br/>feedback"))
        FIE(("Flash<br/>intensity"))
    end

    %% ============================================================
    %% BACKBONE
    %% ============================================================
    subgraph BB["**Backbone**"]
        direction TB
        BL2["Blur Level"]:::software
        CLUST["Clusterize<br/>to 6 welding types"]:::trainedonce
        DEC{{"Routing"}}
        subgraph CNNS["CNNs (1 per welding type)"]

            CNN102["CNN for C102"]:::trainedreal
        end
        LAT(("Latent<br/>Space"))
        THR["Calculation of acceptation thresholds<br/>for all 6 welding types"]:::software
    end

    %% ============================================================
    %% ML COMPONENT - Latent Space Use
    %% ============================================================
    subgraph ML["**ML Component** — Latent Space Use"]
        direction TB
        PRED["Predictor<br/>(MLP regressor)"]:::trainedreal

        ODD["ODD detector<br/>(MLP regressor)"]:::trainedreal
        PROB(("Probabilities<br/>(ok, nok, unk)")):::realdata
        UQK(("UQ KPIs"))
        ANOM(("Anomaly scores:<br/>data drift,<br/>signal artefact")):::augdata
        OODB(("Blur-related<br/>OODD signal"))
        WTYPE(("Welding Type"))
        STH(("Short-term<br/>history"))
        MON["Online monitoring<br/>(multi-timescale)"]:::software
    end

    %% ============================================================
